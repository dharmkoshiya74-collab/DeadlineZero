To eliminate split-brain churn post-partition in Raft Lab 2b, you must enforce the Election Safety Invariant (Section 5.4.1) via strict two-tier log comparison and pre-vote phases before updating local state.

When candidate $j$ requests a vote from node $i$, the candidate is at least as up-to-date as the receiver if and only if:

$$S_j \ge S_i \iff (term_j^{last} > term_i^{last}) \;\lor\; (term_j^{last} == term_i^{last} \;\land\; len(\log_j) \ge len(\log_i))$$

Crucially, if args.Term > rf.currentTerm, step down immediately to Follower, but do not grant the vote unless the candidate's log is strictly monotonic relative to your commit history.

func (rf *Raft) RequestVote(args *RequestVoteArgs, reply *RequestVoteReply) {
    rf.mu.Lock()
    defer rf.mu.Unlock()

    // Rule 1: Reject stale candidates
    if args.Term < rf.currentTerm {
        reply.Term = rf.currentTerm
        reply.VoteGranted = false
        return
    }

    // Rule 2: Rule of Election Safety & Monotonicity
    if args.Term > rf.currentTerm {
        rf.currentTerm = args.Term
        rf.state = Follower
        rf.votedFor = -1
        rf.persist()
    }

    lastLogIndex := len(rf.log) - 1
    lastLogTerm := rf.log[lastLogIndex].Term

    isUpToDate := (args.LastLogTerm > lastLogTerm) ||
        (args.LastLogTerm == lastLogTerm && args.LastLogIndex >= lastLogIndex)

    if (rf.votedFor == -1 || rf.votedFor == args.CandidateId) && isUpToDate {
        rf.votedFor = args.CandidateId
        rf.lastHeartbeat = time.Now()
        rf.persist()
        reply.VoteGranted = true
    } else {
        reply.VoteGranted = false
    }
    reply.Term = rf.currentTerm
}
